cmake_minimum_required(VERSION 3.20)
project(sturgis_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(STURGIS_COVERAGE "Instrument first-party memory worker" ON)
find_package(SQLite3 REQUIRED)
add_executable(sturgis-memory native/memory/main.cpp native/memory/engine.cpp)
target_include_directories(sturgis-memory PRIVATE native/memory/vendor)
target_link_libraries(sturgis-memory PRIVATE SQLite::SQLite3)
target_compile_options(sturgis-memory PRIVATE -Wall -Wextra -Wpedantic)
if(STURGIS_COVERAGE AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(sturgis-memory PRIVATE -O0 -g --coverage)
  target_link_options(sturgis-memory PRIVATE --coverage)
endif()
enable_testing()
